"""Shared observation, worker-process, and result handling for diffusion adapters."""

from __future__ import annotations

from dataclasses import dataclass
import hashlib
import json
import math
import os
from pathlib import Path
import re
import subprocess
import tempfile
import zipfile

import numpy as np

from grasp import PROJECT_ROOT, __version__
from grasp.preprocessing import output_path, validate_T, write_json


def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def canonicalize(poses: np.ndarray, scores: np.ndarray, T_native_tool: np.ndarray,
                 width: float, max_candidates: int, method: str = "graspgenx") -> dict:
    """Preserve raw results; rank and transform candidates into panda_hand_tcp."""
    poses = np.asarray(poses, dtype=np.float32)
    scores = np.asarray(scores, dtype=np.float32)
    if poses.size == 0 and scores.size == 0:
        poses, scores = np.empty((0, 4, 4), np.float32), np.empty(0, np.float32)
    if scores.ndim != 1 or poses.shape != (len(scores), 4, 4):
        raise ValueError("Backend must return poses (N,4,4) and scores (N,)")
    if not np.isfinite(scores).all() or np.any((scores < 0) | (scores > 1)):
        raise ValueError("Native scores must be finite and in [0,1]")
    for pose in poses:
        validate_T(pose)
    T_native_tool = validate_T(T_native_tool)
    indices = np.argsort(-scores, kind="stable")[:max_candidates]
    return {
        "poses": (poses[indices].astype(np.float64) @ T_native_tool).astype(np.float32),
        "scores": scores[indices], "widths": np.full(len(indices), width, np.float32),
        "candidate_ids": np.array([f"{method}:{i:06d}" for i in indices], dtype="U32"),
        "native_indices": indices.astype(np.int64),
        "native_poses": poses, "native_scores": scores, "T_native_tool": T_native_tool,
    }


def load_observation(path: Path, min_points: int) -> tuple[np.ndarray, dict]:
    metadata = json.loads(path.with_name("metadata.json").read_text())
    if not isinstance(metadata, dict):
        raise ValueError("Observation metadata must be a JSON object")
    for field in ("object_id", "view_id", "camera_frame"):
        if not isinstance(metadata.get(field), str) or not metadata[field]:
            raise ValueError(f"Observation metadata requires {field}")
    for field in ("object_id", "view_id"):
        if not re.fullmatch(r"[A-Za-z0-9_-]+", metadata[field]):
            raise ValueError(f"Unsafe observation identifier: {field}")
    if metadata.get("length_unit") != "meter" or metadata.get("schema_version") != 1:
        raise ValueError("Expected schema version 1 observation with meter units")
    with np.load(path, allow_pickle=False) as observation:
        xyz = observation["xyz"]
        mask = observation["point_object_mask"]
        if xyz.ndim != 2 or xyz.shape[1] != 3 or mask.shape != (len(xyz),) or mask.dtype != np.bool_:
            raise ValueError("Expected xyz (N,3) and boolean point_object_mask (N,)")
        points = np.asarray(xyz[mask], dtype=np.float32)
    if len(points) < min_points:
        raise ValueError(f"Object has {len(points)} depth points; at least {min_points} required")
    if not np.isfinite(points).all() or np.any(points[:, 2] <= 0):
        raise ValueError("Object points must be finite and in front of the RGB camera")
    return points, metadata


@dataclass(frozen=True)
class BackendSpec:
    method: str
    toolbox: Path
    python: Path
    worker: Path
    T_native_tool: np.ndarray
    width: float
    native_tool_frame: str
    request: dict
    assets: dict[str, Path]
    environment: dict[str, str]
    redactions: list[tuple[Path, str]]


class IsolatedDiffusionAdapter:
    """Share I/O and execution policy while each toolbox prepares its own backend."""

    def prepare_backend(self) -> BackendSpec:
        raise NotImplementedError

    def generate(self, inputs: list[Path], destination: Path, overwrite: bool = False) -> dict:
        destination = output_path(destination)
        if min(self.num_grasps, self.max_candidates, self.min_object_points) <= 0:
            raise ValueError("Candidate and point counts must be positive")
        if not 0 <= self.seed < 2**32 or not math.isfinite(self.timeout) or self.timeout <= 0:
            raise ValueError("Seed must be in [0,2**32); timeout must be finite and positive")
        spec = self.prepare_backend()
        toolbox, python = spec.toolbox, spec.python
        T_native_tool, width = spec.T_native_tool, spec.width
        if not inputs:
            raise ValueError("No observation.npz inputs found")
        destination.mkdir(parents=True, exist_ok=True)
        configuration = {"gripper": "franka_panda", "planner": "diffusion", "num_grasps": self.num_grasps,
                         "max_candidates": self.max_candidates, "seed": self.seed,
                         "min_object_points": self.min_object_points, "remove_outliers": self.remove_outliers,
                         "grasp_threshold": -1.0, "min_grasps": 1, "max_tries": 1}
        manifest = {"schema_version": 1, "adapter_version": __version__, "method": spec.method,
                    "output_path_base": "manifest_directory", "configuration": configuration, "trials": []}
        redactions = [(str(p), label) for p, label in (
            (toolbox, "<toolbox>"), (python.parent.parent, "<toolbox_environment>"),
            (PROJECT_ROOT, "<project>"), *spec.redactions)]
        redactions.extend((str(path.expanduser().resolve().parent), "<observation>") for path in inputs)
        def redact(error):
            message = str(error)
            for path, label in sorted(redactions, key=lambda item: -len(item[0])):
                message = message.replace(path, label)
            return message
        with tempfile.TemporaryDirectory(prefix=".worker_", dir=destination) as temporary:
            work = Path(temporary)
            jobs, observations, targets, seen = [], [], [], set()
            for path in inputs:
                record = {"observation": path.parent.name, "status": "failed"}
                manifest["trials"].append(record)
                try:
                    points, metadata = load_observation(path, self.min_object_points)
                    key = (metadata["object_id"], metadata["view_id"])
                    record.update(object_id=key[0], view_id=key[1])
                    if key in seen:
                        raise ValueError("Duplicate object/view identifiers in input observations")
                    seen.add(key)
                    target = output_path(destination / key[0] / key[1])
                    if target.exists() and any(target.iterdir()) and not overwrite:
                        raise ValueError("Output exists; choose another output or pass --overwrite")
                    target.mkdir(parents=True, exist_ok=True)
                    if overwrite:
                        output_path(target / "metadata.json").unlink(missing_ok=True)
                    index = len(jobs)
                    input_pc, raw_output = work / f"input_{index}.npz", work / f"raw_{index}.npz"
                    np.savez(input_pc, points=points)
                    jobs.append({"input": input_pc.relative_to(PROJECT_ROOT).as_posix(),
                                 "output": raw_output.relative_to(PROJECT_ROOT).as_posix(),
                                 "trial_index": len(manifest["trials"]) - 1})
                    observations.append((path, metadata, len(points)))
                    targets.append(target)
                except (ValueError, OSError, KeyError, EOFError, zipfile.BadZipFile) as exc:
                    record["error"] = redact(exc)
            if jobs:
                request = {"configuration": configuration, "jobs": jobs,
                           "toolbox_root": os.path.relpath(toolbox, PROJECT_ROOT), **spec.request}
                request_path, response_path = work / "request.json", work / "response.json"
                write_json(request_path, request)
                env = os.environ.copy()
                cache = output_path(PROJECT_ROOT / ".cache" / spec.method)
                for key, directory in {"XDG_CACHE_HOME": "xdg", "HF_HOME": "huggingface", "TORCH_HOME": "torch",
                                       "TORCH_EXTENSIONS_DIR": "extensions", "CUDA_CACHE_PATH": "cuda",
                                       "MPLCONFIGDIR": "matplotlib", "NUMBA_CACHE_DIR": "numba",
                                       "TRITON_CACHE_DIR": "triton", "TMPDIR": "tmp"}.items():
                    folder = output_path(cache / directory)
                    folder.mkdir(parents=True, exist_ok=True)
                    env[key] = str(folder)
                env.update(PYTHONDONTWRITEBYTECODE="1", HF_HUB_OFFLINE="1", **spec.environment)
                command = [str(python), str(spec.worker), str(request_path), str(response_path)]
                try:
                    with (work / "worker.log").open("w") as log:
                        process = subprocess.run(command, cwd=PROJECT_ROOT, env=env, stdout=log,
                                                 stderr=subprocess.STDOUT, timeout=self.timeout, check=False)
                    if not response_path.exists():
                        raise RuntimeError(f"Worker exited with code {process.returncode} without a response; see worker.log")
                    response = json.loads(response_path.read_text())
                except (OSError, subprocess.TimeoutExpired, RuntimeError, ValueError) as exc:
                    response = {"error": redact(exc)}
                finally:
                    log_path = work / "worker.log"
                    output_path(destination / "worker.log").write_text(redact(log_path.read_text(errors="replace")) if log_path.exists() else "")
                results = response.get("results", [])
                provenance = None
                for index, job in enumerate(jobs):
                    record = manifest["trials"][job["trial_index"]]
                    try:
                        result = results[index] if index < len(results) else {"error": response.get("error", "Missing worker result")}
                        if "error" in result:
                            raise RuntimeError(result["error"])
                        with np.load(PROJECT_ROOT / job["output"], allow_pickle=False) as raw:
                            arrays = canonicalize(raw["poses"], raw["scores"], T_native_tool, width, self.max_candidates, spec.method)
                        if provenance is None:
                            revision = subprocess.run(["git", "-C", str(toolbox), "rev-parse", "HEAD"],
                                                      capture_output=True, text=True, check=False).stdout.strip()
                            assets = dict(spec.assets)
                            assets.update({name: (PROJECT_ROOT / path).resolve()
                                           for name, path in response.get("assets", {}).items()})
                            provenance = {"toolbox_revision": revision or None, "runtime": response.get("runtime", {}),
                                          "assets": {name: {"name": asset.name, "sha256": sha256(asset)}
                                                     for name, asset in assets.items()}}
                        source, observation, point_count = observations[index]
                        metadata = {"schema_version": 1, "adapter_version": __version__, "method": spec.method,
                                    "object_id": observation["object_id"], "view_id": observation["view_id"],
                                    "camera_frame": observation["camera_frame"], "tool_frame": "panda_hand_tcp",
                                    "native_tool_frame": spec.native_tool_frame,
                                    "length_unit": "meter", "transform_convention": "T_A_B maps B into A",
                                    "pose_convention": "poses = T_camera_native @ T_native_tool",
                                    "width_source": "configured maximum opening; not predicted",
                                    "observation": {"name": source.name, "sha256": sha256(source),
                                                    "metadata_sha256": sha256(source.with_name("metadata.json"))},
                                    "object_points": point_count, "configuration": configuration,
                                    "inference_seconds": result["inference_seconds"],
                                    "model_load_seconds": response.get("model_load_seconds"),
                                    "native_count": len(arrays["native_scores"]), "candidate_count": len(arrays["scores"]),
                                    "collision_filtering": False, "duplicate_suppression": False, **provenance}
                        with output_path(targets[index] / "grasps.npz").open("wb") as stream:
                            np.savez_compressed(stream, **arrays)
                        write_json(targets[index] / "metadata.json", metadata)
                        record.update(status="ok" if len(arrays["scores"]) else "no_grasps",
                                      output=targets[index].relative_to(destination).as_posix(),
                                      candidate_count=len(arrays["scores"]))
                    except (ValueError, OSError, KeyError, RuntimeError) as exc:
                        record["error"] = redact(exc)
        write_json(destination / "manifest.json", manifest)
        return manifest
